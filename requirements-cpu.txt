#   python -m venv ~/envs/mswegnn          # python must be 3.8-3.11 for these wheels
#   source ~/envs/mswegnn/bin/activate
#   pip install -U pip
#   pip install -r requirements-cpu.txt
#   python -m ipykernel install --user --name mswegnn --display-name "Python (mswegnn)"
